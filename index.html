// ========================================== // 1. 處理前端的 GET 請求 (驗證、取得設定、讀取流水帳、雙層識別) // ========================================== function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 帳號密碼驗證 if (action === 'verify') { const email = e.parameter.email; const code = e.parameter.code; const userSheet = ss.getSheetByName('Users'); const data = userSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === email && String(data[i][1]) === code) { return ContentService.createTextOutput(JSON.stringify({ success: true, name: data[i][3] })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false })) .setMimeType(ContentService.MimeType.JSON); } // 初始化讀取基礎設定 (Config) if (action === 'getConfig') { const configSheet = ss.getSheetByName('Config'); const data = configSheet.getDataRange().getValues(); const configList = []; for (let i = 1; i < data.length; i++) { configList.push({ supplier: data[i][0], name: data[i][1], rule1: data[i][2], rule2: data[i][3], calcType: data[i][4] }); } return ContentService.createTextOutput(JSON.stringify(configList)) .setMimeType(ContentService.MimeType.JSON); } // 讀取記錄列表 (Records) if (action === 'getRecords') { const recSheet = ss.getSheetByName('Records'); const data = recSheet.getDataRange().getValues(); const records = data.slice(1); return ContentService.createTextOutput(JSON.stringify(records)) .setMimeType(ContentService.MimeType.JSON); } // 核心功能：雙層自動識別機制 (優先 Inventory -> 失敗改查 Config 規則) if (action === 'detectFields') { const prefix = e.parameter.prefix; if (!prefix || prefix.length < 3) { return ContentService.createTextOutput(JSON.stringify({ found: false })).setMimeType(ContentService.MimeType.JSON); } // 第一層：優先嘗試比對 Inventory (A) const invSheet = ss.getSheetByName('Inventory'); if (invSheet) { const invData = invSheet.getDataRange().getValues(); for (let i = 1; i < invData.length; i++) { const invId = String(invData[i][0]); const status = String(invData[i][3]).trim(); // 只要編號匹配且狀態不是已使用，即視為有效庫存 if (invId.startsWith(prefix) && status !== '已使用') { return ContentService.createTextOutput(JSON.stringify({ found: true, supplier: invData[i][1], giftName: invData[i][2] })).setMimeType(ContentService.MimeType.JSON); } } } // 第二層：Inventory 沒找到，改以 Config 自動識別規則 (rule2) 識別 const configSheet = ss.getSheetByName('Config'); if (configSheet) { const configData = configSheet.getDataRange().getValues(); for (let i = 1; i < configData.length; i++) { const rule2 = String(configData[i][3]).trim(); if (!rule2 || rule2 === 'N/A') continue; const parts = rule2.split(':'); if (parts[0] === 'START' && prefix.startsWith(parts[1])) { return ContentService.createTextOutput(JSON.stringify({ found: true, supplier: configData[i][0], giftName: configData[i][1] })).setMimeType(ContentService.MimeType.JSON); } if (parts[0] === 'POS') { const startIdx = parseInt(parts[1], 10); const targetStr = parts[2]; if (prefix.substring(startIdx, startIdx + targetStr.length) === targetStr) { return ContentService.createTextOutput(JSON.stringify({ found: true, supplier: configData[i][0], giftName: configData[i][1] })).setMimeType(ContentService.MimeType.JSON); } } } } return ContentService.createTextOutput(JSON.stringify({ found: false })).setMimeType(ContentService.MimeType.JSON); } // 網頁應用程式預設進入點 return HtmlService.createHtmlOutputFromFile('index') .setTitle('Wedding Command Center') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // ========================================== // 2. 處理前端的 POST 請求 (新增、修改、刪除) // ========================================== function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const recSheet = ss.getSheetByName('Records'); const invSheet = ss.getSheetByName('Inventory'); const postData = JSON.parse(e.postData.contents); const action = postData.action; // 正式存檔 if (action === 'insertRecord') { const timestamp = new Date(); recSheet.appendRow([ timestamp, postData.supplier, postData.gift, postData.start, postData.end, postData.count, postData.user, postData.orderNo, '正式存檔', postData.expiry ]); // 同步更新庫存表狀態為已使用並記錄單號 const invData = invSheet.getDataRange().getValues(); for (let i = 1; i < invData.length; i++) { if (String(invData[i][0]) === String(postData.start)) { invSheet.getRange(i + 1, 4).setValue('已使用'); invSheet.getRange(i + 1, 5).setValue(postData.orderNo); invSheet.getRange(i + 1, 7).setValue(timestamp); break; } } } // 編輯儲存 if (action === 'updateRecord') { const rowId = postData.index + 2; recSheet.getRange(rowId, 2).setValue(postData.supplier); recSheet.getRange(rowId, 3).setValue(postData.gift); recSheet.getRange(rowId, 4).setValue(postData.start); recSheet.getRange(rowId, 5).setValue(postData.end); recSheet.getRange(rowId, 6).setValue(postData.count); recSheet.getRange(rowId, 8).setValue(postData.orderNo); } // 標記刪除並將庫存反向歸位 if (action === 'markDeleted') { const rowId = postData.index + 2; const startCode = recSheet.getRange(rowId, 4).getValue(); recSheet.getRange(rowId, 9).setValue('已登記刪除'); const invData = invSheet.getDataRange().getValues(); for (let i = 1; i < invData.length; i++) { if (String(invData[i][0]) === String(startCode)) { invSheet.getRange(i + 1, 4).setValue('未使用'); invSheet.getRange(i + 1, 5).setValue(''); invSheet.getRange(i + 1, 7).setValue(''); break; } } } return ContentService.createTextOutput(JSON.stringify({ result: 'success' })) .setMimeType(ContentService.MimeType.JSON); }
