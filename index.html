// ========================================== // 1. 核心功能：依首碼比對 Inventory 庫存表 // ========================================== function getInventoryMatch(prefix) { if (!prefix || prefix.length < 3) return { found: false }; const ss = SpreadsheetApp.getActiveSpreadsheet(); const invSheet = ss.getSheetByName('Inventory'); if (!invSheet) return { found: false }; const data = invSheet.getDataRange().getValues(); // 欄位索引對照：0:編號, 1:品牌(Supplier), 2:產品(GiftName), 3:狀態 for (let i = 1; i < data.length; i++) { const invId = String(data[i][0]); if (invId.startsWith(prefix) && data[i][3] === '未使用') { return { found: true, supplier: data[i][1], giftName: data[i][2] }; } } return { found: false }; } // ========================================== // 2. 核心功能：正式存檔紀錄並同步反向歸位庫存 // ========================================== function saveRecord(recordData) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const recSheet = ss.getSheetByName('Records'); const invSheet = ss.getSheetByName('Inventory'); const timestamp = new Date(); // 寫入 Records 流水帳 // 欄位結構：Timestamp|Supplier|GiftName|Start|EndCount(此處拆為Start, End, Count)|User|訂單號|操作 recSheet.appendRow([ timestamp, recordData.supplier, recordData.giftName, recordData.start, recordData.end, recordData.count, recordData.user, recordData.orderId, '正式存檔' ]); // 更新 Inventory 庫存狀態 (精確匹配首碼編號) const invData = invSheet.getDataRange().getValues(); for (let i = 1; i < invData.length; i++) { if (String(invData[i][0]).startsWith(recordData.start)) { invSheet.getRange(i + 1, 4).setValue('已使用'); // 狀態 invSheet.getRange(i + 1, 5).setValue(recordData.orderId); // 單號 invSheet.getRange(i + 1, 7).setValue(timestamp); // 使用登記時間 break; } } return true; } // 網頁應用程式進入點 function doGet() { return HtmlService.createHtmlOutputFromFile('index') .setTitle('Wedding Command Center') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); }
